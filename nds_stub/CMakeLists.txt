set(NDS_STUB_DEKO_SHADERS)
function(nds_stub_compile_shader stage source_no_ext output_name)
    set(out_file ${NDS_STUB_SHADER_DIR}/${output_name}.dksh)
    add_custom_command(OUTPUT ${out_file}
        COMMAND ${NDS_STUB_UAM_EXE} -s ${stage} -o ${out_file} ${source_no_ext}.glsl
        DEPENDS ${source_no_ext}.glsl
    )
    list(APPEND NDS_STUB_DEKO_SHADERS ${out_file})
    set(NDS_STUB_DEKO_SHADERS ${NDS_STUB_DEKO_SHADERS} PARENT_SCOPE)
endfunction()

function(nds_stub_compile_shader_instance stage source_no_ext output_name defines)
    foreach(var IN LISTS defines)
        set(${var} 1)
    endforeach()
    set(generated_file ${NDS_STUB_GENERATED_SHADER_DIR}/${output_name}.glsl)
    configure_file(${source_no_ext}.glsl ${generated_file})
    set(out_file ${NDS_STUB_SHADER_DIR}/${output_name}.dksh)
    add_custom_command(OUTPUT ${out_file}
        COMMAND ${NDS_STUB_UAM_EXE} -s ${stage} -o ${out_file} ${generated_file}
        DEPENDS ${generated_file}
    )
    list(APPEND NDS_STUB_DEKO_SHADERS ${out_file})
    set(NDS_STUB_DEKO_SHADERS ${NDS_STUB_DEKO_SHADERS} PARENT_SCOPE)
endfunction()

function(nds_stub_compile_gpu3d_shader_scaled output_name defines)
    foreach(scale RANGE 1 4)
        math(EXPR NDS_DEKO_SCREEN_WIDTH "256 * ${scale}")
        math(EXPR NDS_DEKO_SCREEN_HEIGHT "192 * ${scale}")
        set(NDS_DEKO_TILE_SIZE 8)
        if (scale LESS 3)
            set(NDS_DEKO_WORK_TILE_MULTIPLIER 48)
        elseif (scale EQUAL 3)
            set(NDS_DEKO_WORK_TILE_MULTIPLIER 16)
        else()
            set(NDS_DEKO_WORK_TILE_MULTIPLIER 8)
        endif()
        nds_stub_compile_shader_instance(comp ${ARCDELTA_CORE_SHADER_DIR}/GPU3D_Comp ${output_name}_x${scale} "${defines}")
    endforeach()
    set(NDS_STUB_DEKO_SHADERS ${NDS_STUB_DEKO_SHADERS} PARENT_SCOPE)
endfunction()

function(nds_stub_compile_compose_shader_scaled output_name defines)
    foreach(scale RANGE 1 4)
        set(NDS_DEKO_COMPOSE_SCALE ${scale})
        nds_stub_compile_shader_instance(frag ${ARCDELTA_CORE_SHADER_DIR}/ComposeBGOBJ_fsh ${output_name}_x${scale} "${defines}")
    endforeach()
    set(NDS_STUB_DEKO_SHADERS ${NDS_STUB_DEKO_SHADERS} PARENT_SCOPE)
endfunction()
